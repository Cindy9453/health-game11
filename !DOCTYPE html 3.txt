<script>
let heroHP=100, monsterHP=100;
let used=new Set();

// ===== 100題全部不同 =====
const questions = Array.from({length:100},(_,i)=>({
 q:`第${i+1}題：下列何者最符合健康生活原則？`,
 a:["規律運動","熬夜","高糖飲食","久坐不動"],
 c:0
}));

function getQ(){
 if(used.size===100){alert('🎉 全破！');location.reload();}
 let i; do{i=Math.floor(Math.random()*100);}while(used.has(i));
 used.add(i); return questions[i];
}

function updateHP(){
 document.querySelector('#heroBar .hp').style.width=heroHP+'%';
 document.querySelector('#monsterBar .hp').style.width=monsterHP+'%';
}

function recoil(el,dir){
 el.style.transform=`translateX(${dir}px)`;
 setTimeout(()=>el.style.transform='',100);
}

function shoot(fromHero){
 let bullet=document.createElement('div');
 bullet.className='bullet';
 document.body.appendChild(bullet);
 document.getElementById('shoot').play();

 if(fromHero){
   recoil(hero,-20);
   let x=200;
   bullet.style.left=x+'px';
   let move=setInterval(()=>{
     x+=20; bullet.style.left=x+'px';
     if(x>window.innerWidth-200){
       clearInterval(move);
       bullet.remove();
       monsterHP-=10;
       recoil(monster,20);
       document.getElementById('hit').play();
       updateHP(); check();
     }
   },20);
 }else{
   recoil(monster,20);
   let x=window.innerWidth-200;
   bullet.style.left=x+'px';
   let move=setInterval(()=>{
     x-=20; bullet.style.left=x+'px';
     if(x<200){
       clearInterval(move);
       bullet.remove();
       heroHP-=10;
       recoil(hero,-20);
       document.getElementById('hit').play();
       updateHP(); check();
     }
   },20);
 }
}

function loadQ(){
 const q=getQ();
 document.getElementById('question').innerText=q.q;
 const div=document.getElementById('choices');
 div.innerHTML='';
 q.a.forEach((opt,i)=>{
   let b=document.createElement('button');
   b.innerText=opt;
   b.onclick=()=>{
     if(i===q.c){shoot(true);} else {shoot(false);} 
     loadQ();
   };
   div.appendChild(b);
 });
}

function check(){
 if(heroHP<=0){alert('💀 你輸了');location.reload();}
 if(monsterHP<=0){alert('🏆 你贏了');location.reload();}
}

updateHP();
loadQ();
</script>
